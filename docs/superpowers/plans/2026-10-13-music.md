# Music Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Background music for every screen: synthesized loops by default (offline, no licenses), optional CC0 MP3 tracks, instant stop/resume driven by game state (needed by Statues), and a persisted mute toggle.

**Architecture:** A pure *director* turns the session into a `MusicCue { mood, playing }` and a diff into a `MusicAction`. `main.ts` applies actions to a `HybridMusic` source, which picks a CC0 track for the mood when the manifest lists one and it hasn't failed, else the Web Audio `SynthMusic`. An `AudioEngine` owns the `AudioContext`, master gain and mute. Only the director, manifest parsing and hybrid selection have logic worth unit-testing; the Web Audio adapters are verified in the browser.

**Tech Stack:** TypeScript 5.9, Web Audio API, HTMLAudioElement, Vitest, Vite + vite-plugin-pwa. pnpm.

**Spec:** `docs/superpowers/specs/2026-10-13-small-room-games-music-design.md` (section C).

```ts
import type { Session } from '../app/session'
import type { AnyMiniGame, Mood } from '../core/types'

/** What should be heard right now. `mood: null` = no music (results screen). */
export interface MusicCue {
  readonly mood: Mood | null
  readonly playing: boolean
}

export type MusicAction = { readonly kind: 'none' } | { readonly kind: 'play'; readonly mood: Mood } | { readonly kind: 'pause' }

const DEFAULT_MOOD: Mood = 'upbeat'

/** Pure: derives the music cue from the session screen. */
export function musicCue(session: Session, games: readonly AnyMiniGame[]): MusicCue {
  const { screen } = session
  switch (screen.kind) {
    case 'menu':
      return { mood: 'menu', playing: true }
    case 'countdown':
      return { mood: games[screen.gameIndex].mood ?? DEFAULT_MOOD, playing: true }
    case 'playing': {
      const game = games[screen.gameIndex]
      return { mood: game.mood ?? DEFAULT_MOOD, playing: game.status(screen.state).music !== 'off' }
    }
    case 'gameOver':
      return { mood: null, playing: false }
  }
}

function audible(cue: MusicCue | null): cue is MusicCue & { mood: Mood } {
  return cue !== null && cue.playing && cue.mood !== null
}

/** Pure: what to tell the music source so it goes from `previous` to `next`. */
export function musicAction(previous: MusicCue | null, next: MusicCue): MusicAction {
  if (audible(next)) {
    return audible(previous) && previous.mood === next.mood ? { kind: 'none' } : { kind: 'play', mood: next.mood }
  }
  return audible(previous) ? { kind: 'pause' } : { kind: 'none' }
}
```

```ts
import type { Mood } from '../core/types'

export interface Track {
  readonly mood: Mood
  /** File name inside public/music/. */
  readonly file: string
}

export interface TrackManifest {
  readonly tracks: readonly Track[]
}

export const EMPTY_MANIFEST: TrackManifest = { tracks: [] }

const MOODS: readonly Mood[] = ['menu', 'upbeat', 'calm']
/** Plain local file names only: no paths, no URLs. */
const SAFE_FILE = /^[a-z0-9][a-z0-9-]*\.mp3$/

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value)
}

function toTrack(value: unknown): Track[] {
  if (!isRecord(value)) return []
  const { mood, file } = value
  if (typeof mood !== 'string' || !(MOODS as readonly string[]).includes(mood)) return []
  if (typeof file !== 'string' || !SAFE_FILE.test(file)) return []
  return [{ mood: mood as Mood, file }]
}

/** Validates untrusted JSON from public/music/manifest.json; invalid entries are dropped. */
export function parseManifest(raw: unknown): TrackManifest {
  if (!isRecord(raw) || !Array.isArray(raw.tracks)) return EMPTY_MANIFEST
  return { tracks: raw.tracks.flatMap(toTrack) }
}

/** URL of the first track for `mood`, or null to fall back to synthesized music. */
export function trackUrl(manifest: TrackManifest, mood: Mood, baseUrl: string): string | null {
  const track = manifest.tracks.find((t) => t.mood === mood)
  return track ? `${baseUrl}music/${track.file}` : null
}
```

```ts
import type { Mood } from '../core/types'
import { EMPTY_MANIFEST, type TrackManifest, trackUrl } from './manifest'

/** Something that can loop music for a mood. `play` on the current mood resumes it. */
export interface MusicSource {
  play(mood: Mood): void
  /** Must silence within ~50 ms (Statues depends on it). */
  pause(): void
}

export interface TrackPlayer {
  play(mood: Mood, url: string): void
  pause(): void
}

/**
 * Plays the CC0 track listed for a mood, or synthesized music when there is
 * none or it failed to load. A failed mood stays on synth for the session.
 */
export class HybridMusic implements MusicSource {
  private manifest: TrackManifest = EMPTY_MANIFEST
  private readonly failed = new Set<Mood>()
  private current: Mood | null = null
  private playing = false

  constructor(
    private readonly synth: MusicSource,
    private readonly tracks: TrackPlayer,
    private readonly baseUrl: string,
  ) {}

  setManifest(manifest: TrackManifest): void {
    this.manifest = manifest
  }

  play(mood: Mood): void {
    this.current = mood
    this.playing = true
    const url = this.failed.has(mood) ? null : trackUrl(this.manifest, mood, this.baseUrl)
    if (url) {
      this.synth.pause()
      this.tracks.play(mood, url)
    } else {
      this.tracks.pause()
      this.synth.play(mood)
    }
  }

  pause(): void {
    this.playing = false
    this.synth.pause()
    this.tracks.pause()
  }

  /** Called by the track player when a file can't be fetched or decoded. */
  trackFailed(mood: Mood): void {
    this.failed.add(mood)
    if (this.playing && this.current === mood) this.play(mood)
  }
}
```

```ts
const MUTED_KEY = 'game-cam.muted'
/** Music sits under the game: loud enough to dance to, quiet enough to talk over. */
const MASTER_VOLUME = 0.5
const MUTE_RAMP_S = 0.02

/** Owns the AudioContext. Browsers only allow starting it from a user gesture: call unlock() in a click handler. */
export class AudioEngine {
  private ctx: AudioContext | null = null
  private master: GainNode | null = null
  private isMuted: boolean

  constructor(private readonly storage: Storage = localStorage) {
    this.isMuted = storage.getItem(MUTED_KEY) === '1'
  }

  /** Creates (or resumes) the context. Safe to call more than once. */
  unlock(): void {
    if (this.ctx) {
      void this.ctx.resume()
      return
    }
    const ctx = new AudioContext()
    const master = ctx.createGain()
    master.gain.value = this.isMuted ? 0 : MASTER_VOLUME
    master.connect(ctx.destination)
    this.ctx = ctx
    this.master = master
    // Background tabs throttle timers, which would garble the sequencer: suspend instead.
    document.addEventListener('visibilitychange', () => {
      void (document.hidden ? ctx.suspend() : ctx.resume())
    })
  }

  /** Context and output bus, or null before unlock(). */
  output(): { ctx: AudioContext; bus: AudioNode } | null {
    return this.ctx && this.master ? { ctx: this.ctx, bus: this.master } : null
  }

  muted(): boolean {
    return this.isMuted
  }

  setMuted(muted: boolean): void {
    this.isMuted = muted
    this.storage.setItem(MUTED_KEY, muted ? '1' : '0')
    if (this.ctx && this.master) {
      this.master.gain.setTargetAtTime(muted ? 0 : MASTER_VOLUME, this.ctx.currentTime, MUTE_RAMP_S)
    }
  }
}
```

```ts
import type { Mood } from '../core/types'
import type { AudioEngine } from './engine'
import type { MusicSource } from './music'

/** Four bars of sixteenth notes. */
const STEPS = 64
const STEPS_PER_BAR = 16
const LOOKAHEAD_S = 0.12
const TICK_MS = 25
const FADE_S = 0.03

type Chord = readonly [number, number, number]

interface Pattern {
  readonly bpm: number
  /** One chord per bar (MIDI notes). */
  readonly chords: readonly [Chord, Chord, Chord, Chord]
  readonly lead: OscillatorType
  /** Bass notes per bar: 2 = half notes, 8 = eighth notes. */
  readonly bassPerBar: number
  /** Sixteenth-note steps (within a bar) that get a kick / a hi-hat. */
  readonly kicks: readonly number[]
  readonly hats: readonly number[]
}

const PATTERNS: Readonly<Record<Mood, Pattern>> = {
  menu: {
    bpm: 96,
    chords: [[57, 60, 64], [53, 57, 60], [60, 64, 67], [55, 59, 62]], // Am F C G
    lead: 'triangle',
    bassPerBar: 2,
    kicks: [0, 8],
    hats: [4, 12],
  },
  upbeat: {
    bpm: 128,
    chords: [[60, 64, 67], [55, 59, 62], [57, 60, 64], [53, 57, 60]], // C G Am F
    lead: 'square',
    bassPerBar: 8,
    kicks: [0, 4, 8, 12],
    hats: [2, 6, 10, 14],
  },
  calm: {
    bpm: 84,
    chords: [[62, 65, 69], [58, 62, 65], [65, 69, 72], [60, 64, 67]], // Dm Bb F C
    lead: 'sine',
    bassPerBar: 2,
    kicks: [0],
    hats: [],
  },
}

/** Chord tone order for the eighth-note arpeggio. */
const ARP = [0, 1, 2, 1] as const

const frequency = (midi: number): number => 440 * 2 ** ((midi - 69) / 12)

/** Loops a 4-bar chiptune pattern per mood with a lookahead scheduler. */
export class SynthMusic implements MusicSource {
  private bus: GainNode | null = null
  private noise: AudioBuffer | null = null
  private timer: number | undefined
  private mood: Mood | null = null
  private step = 0
  private nextTime = 0

  constructor(private readonly engine: AudioEngine) {}

  play(mood: Mood): void {
    const out = this.engine.output()
    if (!out) return
    const bus = this.ensureBus(out.ctx, out.bus)
    if (mood !== this.mood) {
      this.mood = mood
      this.step = 0
    }
    bus.gain.cancelScheduledValues(out.ctx.currentTime)
    bus.gain.setTargetAtTime(1, out.ctx.currentTime, FADE_S / 3)
    if (this.timer === undefined) {
      this.nextTime = out.ctx.currentTime + 0.05
      this.timer = window.setInterval(() => this.tick(out.ctx), TICK_MS)
    }
  }

  pause(): void {
    window.clearInterval(this.timer)
    this.timer = undefined
    const out = this.engine.output()
    if (!out || !this.bus) return
    this.bus.gain.cancelScheduledValues(out.ctx.currentTime)
    this.bus.gain.setTargetAtTime(0, out.ctx.currentTime, FADE_S / 3)
  }

  private ensureBus(ctx: AudioContext, destination: AudioNode): GainNode {
    if (this.bus) return this.bus
    const bus = ctx.createGain()
    bus.gain.value = 0
    bus.connect(destination)
    const noise = ctx.createBuffer(1, Math.floor(ctx.sampleRate * 0.05), ctx.sampleRate)
    const data = noise.getChannelData(0)
    for (let i = 0; i < data.length; i += 1) data[i] = Math.random() * 2 - 1
    this.bus = bus
    this.noise = noise
    return bus
  }

  private tick(ctx: AudioContext): void {
    if (!this.mood) return
    const pattern = PATTERNS[this.mood]
    const stepSeconds = 60 / pattern.bpm / 4
    while (this.nextTime < ctx.currentTime + LOOKAHEAD_S) {
      this.schedule(ctx, pattern, this.step, this.nextTime, stepSeconds)
      this.nextTime += stepSeconds
      this.step = (this.step + 1) % STEPS
    }
  }

  private schedule(ctx: AudioContext, p: Pattern, step: number, time: number, stepSeconds: number): void {
    const chord = p.chords[Math.floor(step / STEPS_PER_BAR)]
    const inBar = step % STEPS_PER_BAR
    if (inBar % 2 === 0) {
      const tone = chord[ARP[(inBar / 2) % ARP.length]]
      this.tone(ctx, frequency(tone + 12), time, stepSeconds * 1.8, p.lead, 0.07)
    }
    const bassEvery = STEPS_PER_BAR / p.bassPerBar
    if (inBar % bassEvery === 0) this.tone(ctx, frequency(chord[0] - 24), time, stepSeconds * bassEvery * 0.9, 'triangle', 0.22)
    if (p.kicks.includes(inBar)) this.kick(ctx, time)
    if (p.hats.includes(inBar)) this.hat(ctx, time)
  }

  private tone(ctx: AudioContext, freq: number, time: number, duration: number, type: OscillatorType, volume: number): void {
    if (!this.bus) return
    const osc = ctx.createOscillator()
    const gain = ctx.createGain()
    osc.type = type
    osc.frequency.value = freq
    gain.gain.setValueAtTime(0.0001, time)
    gain.gain.exponentialRampToValueAtTime(volume, time + 0.008)
    gain.gain.exponentialRampToValueAtTime(0.0001, time + duration)
    osc.connect(gain).connect(this.bus)
    osc.start(time)
    osc.stop(time + duration + 0.02)
  }

  private kick(ctx: AudioContext, time: number): void {
    if (!this.bus) return
    const osc = ctx.createOscillator()
    const gain = ctx.createGain()
    osc.frequency.setValueAtTime(150, time)
    osc.frequency.exponentialRampToValueAtTime(40, time + 0.12)
    gain.gain.setValueAtTime(0.6, time)
    gain.gain.exponentialRampToValueAtTime(0.0001, time + 0.15)
    osc.connect(gain).connect(this.bus)
    osc.start(time)
    osc.stop(time + 0.16)
  }

  private hat(ctx: AudioContext, time: number): void {
    if (!this.bus || !this.noise) return
    const source = ctx.createBufferSource()
    const filter = ctx.createBiquadFilter()
    const gain = ctx.createGain()
    source.buffer = this.noise
    filter.type = 'highpass'
    filter.frequency.value = 7000
    gain.gain.setValueAtTime(0.12, time)
    gain.gain.exponentialRampToValueAtTime(0.0001, time + 0.04)
    source.connect(filter).connect(gain).connect(this.bus)
    source.start(time)
  }
}
```

```ts
import type { Mood } from '../core/types'
import type { AudioEngine } from './engine'
import type { TrackPlayer } from './music'

const FADE_S = 0.03
const PAUSE_DELAY_MS = 60

/** Loops one `<audio>` element per mood, routed through the engine so mute applies. */
export class TrackMusic implements TrackPlayer {
  private readonly elements = new Map<Mood, HTMLAudioElement>()
  private bus: GainNode | null = null
  private active: HTMLAudioElement | null = null
  private pauseTimer: number | undefined

  constructor(
    private readonly engine: AudioEngine,
    private readonly onError: (mood: Mood) => void,
  ) {}

  play(mood: Mood, url: string): void {
    const out = this.engine.output()
    if (!out) return
    window.clearTimeout(this.pauseTimer)
    const element = this.elementFor(mood, url, out.ctx, out.bus)
    if (this.active && this.active !== element) this.active.pause()
    this.active = element
    this.bus!.gain.cancelScheduledValues(out.ctx.currentTime)
    this.bus!.gain.setTargetAtTime(1, out.ctx.currentTime, FADE_S / 3)
    element.play().catch((error: unknown) => {
      console.warn(`Music track for "${mood}" could not play; using synthesized music`, error)
      this.onError(mood)
    })
  }

  pause(): void {
    const out = this.engine.output()
    if (!out || !this.bus || !this.active) return
    this.bus.gain.cancelScheduledValues(out.ctx.currentTime)
    this.bus.gain.setTargetAtTime(0, out.ctx.currentTime, FADE_S / 3)
    const element = this.active
    window.clearTimeout(this.pauseTimer)
    this.pauseTimer = window.setTimeout(() => element.pause(), PAUSE_DELAY_MS)
  }

  private elementFor(mood: Mood, url: string, ctx: AudioContext, destination: AudioNode): HTMLAudioElement {
    const existing = this.elements.get(mood)
    if (existing) return existing
    if (!this.bus) {
      this.bus = ctx.createGain()
      this.bus.gain.value = 0
      this.bus.connect(destination)
    }
    const element = new Audio(url)
    element.loop = true
    element.preload = 'auto'
    element.addEventListener('error', () => {
      console.warn(`Music track for "${mood}" failed to load; using synthesized music`, element.error)
      this.onError(mood)
    })
    ctx.createMediaElementSource(element).connect(this.bus)
    this.elements.set(mood, element)
    return element
  }
}
```
